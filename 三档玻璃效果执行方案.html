一、三档定义与总览

档位用户感知localStorageDOM 门控渲染方式默认
关闭 off 实色卡片，与现状完全一致 engchain-glass = 'off' <html> 无 data-glass app.css 既有实色令牌 是（保守默认）
标准 standard 毛玻璃卡片 + 边缘高光，全站统一 engchain-glass = 'standard' data-glass="on" glass-upgrade.css 的 CSS backdrop-filter
增强 enhanced 标准玻璃基础上，精选英雄面有真实折射 engchain-glass = 'enhanced' data-glass="on" data-glass-level="enhanced" CSS 玻璃全量 + WebGL 液态折射（仅英雄面）
💡 为什么默认关闭？ 批量挂载 glass-upgrade.css 本身是一次全站视觉变更，不应在用户首次升级时静默打开。由用户在设置页主动选择，后续灰度稳定后再考虑调整默认值。首次访问用户视觉与现状完全一致，零感知。
⚠️ 方案核实中发现的两个既有问题（执行时顺带修复）
  1. glass-upgrade.css 暗色选择器 bug：L46 现写法 [data-theme="dark"] html[data-glass="on"] 永远不命中（data-theme 直接挂在 html 上，正确应为 html[data-theme="dark"][data-glass="on"]）。当前暗色主题下的玻璃阴影微调实际未生效。
  2. engchain:theme 不是 CustomEvent：common.js L1247 的 engchain:theme 实际是 postMessage 消息类型（宿主预览 iframe 下发主题），theme.set() 内部不 dispatch 任何事件。因此 glass↔theme 联动改用 MutationObserver 监听 data-theme 属性，零侵入。

三、实现方案（具体文件与改动点）

3.1 状态管理（common.js 新增 glass 对象）

在 js/common.js L398（theme 对象紧后）新增 glass 对象，完全镜像 theme 的 API 风格，并在 L1135 return 对象中导出为 UI.glass。

/* 玻璃效果三档：off | standard | enhanced
   off = 无 data-glass；standard = data-glass="on"（纯CSS）；
   enhanced = data-glass="on" + data-glass-level="enhanced"（CSS + WebGL hero） */
const GLASS_TIERS = ['off', 'standard', 'enhanced'];
const glass = {
  TIERS: GLASS_TIERS,
  get() {
    var t = document.documentElement.getAttribute('data-glass-level');
    var on = document.documentElement.hasAttribute('data-glass');
    if (!on) return 'off';
    return t === 'enhanced' ? 'enhanced' : 'standard';
  },
  set(tier) {
    if (GLASS_TIERS.indexOf(tier) === -1) tier = 'off';
    var el = document.documentElement;
    if (tier === 'off') {
      el.removeAttribute('data-glass');
      el.removeAttribute('data-glass-level');
    } else {
      el.setAttribute('data-glass', 'on');
      if (tier === 'enhanced') el.setAttribute('data-glass-level', 'enhanced');
      else el.removeAttribute('data-glass-level');
    }
    try { localStorage.setItem('engchain-glass', tier); } catch (e) {}
    // 通知增强档 loader：切到 enhanced 时建实例，切走时销毁
    try { window.dispatchEvent(new CustomEvent('engchain:glass', { detail: { tier: tier } })); } catch (e) {}
  },
  next() {
    var i = GLASS_TIERS.indexOf(this.get());
    return this.set(GLASS_TIERS[(i + 1) % GLASS_TIERS.length]);
  },
};
🔗 与 theme 的联动方式 经核实，common.js L1247 的 engchain:theme 是 postMessage 消息（宿主预览 iframe 下发主题），不是 DOM CustomEvent，theme.set() 内部也不 dispatch 事件。因此增强档 loader 内部使用 MutationObserver 监听 <html data-theme> 属性变化，不修改 common.js 的 theme 对象，零侵入。

3.2 持久化与重置（databus.js KEEP 白名单）

js/databus.js L232 当前为：

var KEEP = ['engchain-theme', 'engchain-sidenav'];

3.3 防闪烁（FOUC）内联脚本

镜像现有 data-theme-fix 的 IIFE 风格，在每个页面 <head> 最顶部、theme-fix 脚本紧后插入：

<script data-glass-fix="1">(function(){try{var g=localStorage.getItem('engchain-glass');var el=document.documentElement;if(g==='standard'){el.setAttribute('data-glass','on');}else if(g==='enhanced'){el.setAttribute('data-glass','on');el.setAttribute('data-glass-level','enhanced');}}catch(e){}})();</script>

head 脚本顺序：

  1. <script data-theme-fix="1">（已有，必须第一）
  2. <script data-glass-fix="1">（新增，紧随其后）
  3. <script data-engchain-basefix="2">（已有，保持原位）

批量挂载方式：约 115 个 HTML 文件，用一次性 PowerShell 脚本做幂等注入——递归扫描 .html（排除 node_modules/vendor），已含 data-glass-fix 的跳过，在 data-theme-fix 闭合后插入。相对路径从该文件已有的 app.css link 反推，根目录页用 css/，内页用 ../../css/。执行前先 dry-run 输出待改文件清单，人工抽样确认后再写盘。

3.5.2 新建 js/glass-enhanced.js

以 <script type="module"> 加载（module 默认 defer，天然在 DOMContentLoaded 后执行）。核心职责：

职责实现方式
实例白名单只对标注 data-glass-hero 属性的元素创建 LiquidGlass 实例
实例上限单页最多 4 个（Dock AI球已占1个，留余量；超出的 hero 只享受 CSS 玻璃）
档位响应监听 engchain:glass CustomEvent，切到 enhanced 时 build，切走时 destroyAll
主题联动MutationObserver 监听 data-theme，切换 tintTone（light↔dark），调实例 update() 或重建
降级prefers-reduced-motion / prefers-reduced-transparency 时不创建实例，回退标准档；WebGL2 不可用时回退标准档
性能预算maxDpr: 1.5（默认2，移动端平衡清晰度与填充率）；requestIdleCallback 创建（超时1200ms兜底）
销毁pagehide 事件 destroyAll；每个实例 try/catch 包裹，单个失败不影响其他
// js/glass-enhanced.js 骨架（type=module）
import { LiquidGlass } from '../vendor/liquid-glass/src/index.js';

const HERO_SELECTOR = '[data-glass-hero]';
const MAX_INSTANCES = 4;
let instances = [];

function tier() { /* 读 html 属性判断 enhanced */ }
function reducedTransparency() { /* matchMedia 检测 */ }
function webglOK() { /* 创建 canvas 检测 webgl2 */ }

function destroyAll() {
  instances.forEach(({ glass }) => { try { glass.destroy(); } catch (e) {} });
  instances = [];
}

function build() {
  destroyAll();
  if (tier() !== 'enhanced') return;
  if (reducedTransparency() || !webglOK()) return; // 回退 standard CSS 玻璃
  var tone = document.documentElement.dataset.theme === 'dark' ? 'dark' : 'light';
  var els = Array.prototype.slice.call(
    document.querySelectorAll(HERO_SELECTOR), 0, MAX_INSTANCES
  );
  els.forEach(function (el) {
    try {
      var g = new LiquidGlass(el, {
        tint: 0.6, tintTone: tone, frost: 0.85,
        backdrop: 'auto', maxDpr: 1.5,
        respectReducedTransparency: true,
      });
      instances.push({ el: el, glass: g });
    } catch (e) { /* 单个失败不影响其他 */ }
  });
}

window.addEventListener('engchain:glass', build);
new MutationObserver(function () { /* 主题切换时 update tintTone */ })
  .observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });
if ('requestIdleCallback' in window) requestIdleCallback(build, { timeout: 1200 });
else window.addEventListener('load', function () { setTimeout(build, 200); });
window.addEventListener('pagehide', destroyAll);
