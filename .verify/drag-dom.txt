function drag(handleEl, fromY, toY, steps){
  return new Promise(function(res){
    var rect = handleEl.getBoundingClientRect();
    var x = rect.left + rect.width / 2;
    var y0 = (fromY === undefined) ? (rect.top + rect.height / 2) : fromY;
    var n = steps || 10;
    function fire(type, yy){
      var ev = new MouseEvent(type, { bubbles: true, cancelable: true, clientX: x, clientY: yy, button: 0 });
      (type === 'mousemove' || type === 'mouseup' ? window : handleEl).dispatchEvent(ev);
    }
    fire('mousedown', y0);
    var i = 0;
    (function step(){
      if (i > n) { fire('mouseup', toY); setTimeout(res, 500); return; }
      fire('mousemove', y0 + (toY - y0) * i / n);
      i++; setTimeout(step, 25);
    })();
  });
}
