function fireMouse(el, type, x, y){
  el.dispatchEvent(new MouseEvent(type, {bubbles:true, cancelable:true, clientX:x, clientY:y, button:0}));

function dragOnHandle(handle, y0, y1){
  return new Promise(function(res){
    var r = handle.getBoundingClientRect();
    var x = r.left + r.width/2;
    fireMouse(handle,'mousedown',x,y0);
    var n=10, i=0;
    (function step(){
      if(i>n){ fireMouse(window,'mouseup',x,y1); setTimeout(res, 500); return; }
      var yy = y0 + (y1-y0)*i/n;
      fireMouse(window,'mousemove',x,yy);
      i++; setTimeout(step, 15);
    })();
  });
}
