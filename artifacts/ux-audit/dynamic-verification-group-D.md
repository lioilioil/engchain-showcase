### 状态切换JS代码模板
```javascript
// 游客
localStorage.setItem('engchain-state', JSON.stringify({user:'',company:'',status:'guest',member:false,loggedIn:false}));

// 实名用户（破冰期）
localStorage.setItem('engchain-state', JSON.stringify({user:'陈建国',company:'××建设',status:'realname',member:false,loggedIn:true}));
AuthStore.write({realname:{ok:true,ts:Date.now()},enterprise:{ok:false,expireAt:0},qual:{ok:false,list:[]}});
EntryStore.write({type:null,status:null,active:false});
ModeStore.switch('breakin');

// 入驻-建筑（破冰期）
AuthStore.write({realname:{ok:true,ts:Date.now()},enterprise:{ok:true,expireAt:Date.now()+365*864e5},qual:{ok:true,list:['建筑资质']}});
EntryStore.write({type:'construction',status:'active',active:true,paidAt:Date.now(),expireAt:Date.now()+365*864e5});
localStorage.setItem('engchain-state', JSON.stringify({user:'陈建国',company:'××建设',status:'resident',member:true,loggedIn:true}));
```
